## 核心模块说明

### 1. cmd/server/main.go

程序入口，负责：
- 加载配置
- 初始化各组件
- 启动 HTTP 服务

```go
func main() {
    // 1. 加载配置
    cfg := config.Load()

    // 2. 初始化数据库
    db := initDB(cfg)

    // 3. 初始化 SSH 执行器
    sshExecutor := ssh.NewExecutor(cfg.SSH)

    // 4. 初始化 Tool 注册中心
    toolRegistry := tool.NewRegistry()
    tool.RegisterBuiltinTools(toolRegistry)
    tool.LoadScriptTools(cfg.ScriptDir, toolRegistry)

    // 5. 初始化 LLM 客户端
    llmClient := llm.NewOpenAIClient(cfg.LLM)

    // 6. 初始化 Agent
    agent := agent.New(llmClient, toolRegistry, sshExecutor)

    // 7. 初始化 HTTP 服务
    router := api.NewRouter(agent, db)

    // 8. 启动服务
    router.Run(cfg.Server.Addr)
}
```

### 2. internal/agent/agent.go

Agent 核心循环：

```go
type Agent struct {
    llm      llm.Client
    tools    *tool.Registry
    ssh      ssh.Executor
    maxLoops int
}

func (a *Agent) Chat(ctx context.Context, session *Session, userMsg string) (*Response, error) {
    // 1. 添加用户消息到历史
    session.AddMessage(RoleUser, userMsg)

    // 2. 构造 Prompt
    prompt := a.buildPrompt(session)

    // 3. Agent 循环
    for i := 0; i < a.maxLoops; i++ {
        // 调用 LLM
        llmResp, err := a.llm.Chat(ctx, prompt)
        if err != nil {
            return nil, err
        }

        // 解析 LLM 输出
        parsed := a.parseResponse(llmResp)

        // 如果是工具调用
        if parsed.ToolCall != nil {
            // 执行工具
            result := a.executeTool(ctx, parsed.ToolCall)

            // 添加到历史
            session.AddToolCall(parsed.ToolCall, result)

            // 继续循环
            prompt = a.buildPrompt(session)
            continue
        }

        // 如果是最终回复
        session.AddMessage(RoleAssistant, parsed.Content)
        return &Response{Content: parsed.Content}, nil
    }

    return nil, errors.New("达到最大循环次数")
}
```

### 5. internal/api/router.go

HTTP 路由：

```go
func NewRouter(agent *agent.Agent, db *gorm.DB) *gin.Engine {
    r := gin.Default()

    // 中间件
    r.Use(middleware.Logger())
    r.Use(middleware.CORS())

    // API 路由
    api := r.Group("/api")
    {
        // 对话
        api.POST("/chat", handler.Chat(agent))
        api.GET("/chat/history", handler.ChatHistory())
        api.GET("/chat/sessions", handler.Sessions())

        // 主机
        api.GET("/hosts", handler.ListHosts())
        api.POST("/hosts", handler.CreateHost())
        api.PUT("/hosts/:id", handler.UpdateHost())
        api.DELETE("/hosts/:id", handler.DeleteHost())

        // 工具
        api.GET("/tools", handler.ListTools())

        // 脚本
        api.GET("/scripts", handler.ListScripts())
        api.POST("/scripts", handler.UploadScript())

        // 系统
        api.GET("/health", handler.Health())
    }

    // 静态文件 (前端)
    r.Static("/", "./web/dist")

    return r
}
```

## Makefile

```makefile
.PHONY: build run test clean

# 构建
build:
	go build -o bin/aiops ./cmd/server

# 运行
run:
	go run ./cmd/server

# 测试
test:
	go test -v ./...

# 构建前端
build-web:
	cd web && npm install && npm run build

# 构建全部
build-all: build-web build

# 清理
clean:
	rm -rf bin/
	rm -rf web/dist/

# Docker 构建
docker-build:
	docker build -t aiops:latest .

# Docker 运行
docker-run:
	docker run -p 8080:8080 -v $(PWD)/config.yaml:/app/config.yaml aiops:latest
```

## Dockerfile

```dockerfile
# 构建阶段 - 后端
FROM golang:1.21-alpine AS backend-builder
WORKDIR /app
COPY go.mod go.sum ./
RUN go mod download
COPY . .
RUN go build -o aiops ./cmd/server

# 构建阶段 - 前端
FROM node:18-alpine AS frontend-builder
WORKDIR /app
COPY web/package*.json ./
RUN npm install
COPY web/ .
RUN npm run build

# 运行阶段
FROM alpine:latest
WORKDIR /app
COPY --from=backend-builder /app/aiops .
COPY --from=frontend-builder /app/dist ./web/dist
COPY config.yaml .
EXPOSE 8080
CMD ["./aiops"]
```
