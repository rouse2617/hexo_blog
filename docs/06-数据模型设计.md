## 模型定义

### Host 主机

```go
type Host struct {
    ID        string    `json:"id" gorm:"primaryKey"`
    Name      string    `json:"name" gorm:"uniqueIndex;not null"`      // 显示名称
    IP        string    `json:"ip" gorm:"not null"`                    // IP 地址
    Port      int       `json:"port" gorm:"default:22"`                // SSH 端口
    User      string    `json:"user" gorm:"default:root"`              // SSH 用户
    Group     string    `json:"group" gorm:"index"`                    // 分组
    Tags      []string  `json:"tags" gorm:"serializer:json"`           // 标签
    AuthType  string    `json:"auth_type" gorm:"default:key"`          // password / key
    Password  string    `json:"-" gorm:""`                             // 密码 (加密存储)
    KeyPath   string    `json:"key_path" gorm:""`                      // 密钥路径
    Status    string    `json:"status" gorm:"default:unknown"`         // online / offline / unknown
    LastCheck time.Time `json:"last_check"`                            // 最后检查时间
    CreatedAt time.Time `json:"created_at"`
    UpdatedAt time.Time `json:"updated_at"`
}

// 状态常量
const (
    HostStatusOnline  = "online"
    HostStatusOffline = "offline"
    HostStatusUnknown = "unknown"
)

// 认证类型
const (
    AuthTypePassword = "password"
    AuthTypeKey      = "key"
)
```

### Message 对话消息

```go
type Message struct {
    ID        string     `json:"id" gorm:"primaryKey"`
    SessionID string     `json:"session_id" gorm:"index;not null"`
    Role      string     `json:"role"`                          // user / assistant / tool
    Content   string     `json:"content"`                       // 消息内容
    ToolCalls []ToolCall `json:"tool_calls" gorm:"serializer:json"` // 工具调用记录
    CreatedAt time.Time  `json:"created_at"`

// ToolCall 工具调用记录
type ToolCall struct {
    Tool   string         `json:"tool"`
    Params map[string]any `json:"params"`
    Result string         `json:"result"`
    Error  string         `json:"error,omitempty"`
}

// 角色常量
const (
    RoleUser      = "user"
    RoleAssistant = "assistant"
    RoleTool      = "tool"

```

## 数据库初始化

```go
func InitDB(dbPath string) (*gorm.DB, error) {
    db, err := gorm.Open(sqlite.Open(dbPath), &gorm.Config{})
    if err != nil {
        return nil, err
    }

    // 自动迁移
    err = db.AutoMigrate(
        &Host{},
        &Group{},
        &Session{},
        &Message{},
        &Script{},
        &AuditLog{},
        &Task{},
        &Config{},
    )
    if err != nil {
        return nil, err
    }

    return db, nil
}
```

## Repository 层

```go
// HostRepository 主机仓库
type HostRepository interface {
    Create(host *Host) error
    Update(host *Host) error
    Delete(id string) error
    GetByID(id string) (*Host, error)
    GetByName(name string) (*Host, error)
    List(filter HostFilter) ([]*Host, error)
    UpdateStatus(id string, status string) error
}

type HostFilter struct {
    Group   string
    Keyword string
    Status  string
    Tags    []string
}

// SessionRepository 会话仓库
type SessionRepository interface {
    Create(session *Session) error
    GetByID(id string) (*Session, error)
    List(limit int) ([]*Session, error)
    Delete(id string) error
    AddMessage(sessionID string, msg *Message) error
    GetMessages(sessionID string) ([]*Message, error)
}

// AuditLogRepository 审计日志仓库
type AuditLogRepository interface {
    Create(log *AuditLog) error
    List(filter AuditFilter) ([]*AuditLog, error)
}

type AuditFilter struct {
    SessionID string
    Host      string
    User      string
    StartTime time.Time
    EndTime   time.Time
}
```

## 数据清理策略

```go
// 定期清理旧数据
type CleanupConfig struct {
    SessionRetentionDays  int // 会话保留天数
    AuditLogRetentionDays int // 审计日志保留天数
    TaskRetentionDays     int // 任务记录保留天数
}

func CleanupOldData(db *gorm.DB, config CleanupConfig) error {
    now := time.Now()

    // 清理旧会话
    sessionCutoff := now.AddDate(0, 0, -config.SessionRetentionDays)
    db.Where("updated_at < ?", sessionCutoff).Delete(&Session{})
    db.Where("session_id NOT IN (SELECT id FROM sessions)").Delete(&Message{})

    // 清理旧审计日志
    auditCutoff := now.AddDate(0, 0, -config.AuditLogRetentionDays)
    db.Where("created_at < ?", auditCutoff).Delete(&AuditLog{})

    // 清理旧任务
    taskCutoff := now.AddDate(0, 0, -config.TaskRetentionDays)
    db.Where("created_at < ? AND status IN ?", taskCutoff,
        []string{TaskStatusCompleted, TaskStatusFailed, TaskStatusCancelled}).Delete(&Task{})

    return nil
}
```
