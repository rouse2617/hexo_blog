## 状态管理 (Pinia)

```typescript
// stores/chat.ts
export const useChatStore = defineStore('chat', {
  state: () => ({
    sessions: [] as Session[],
    currentSessionId: null as string | null,
    messages: [] as Message[],
    selectedHosts: [] as string[],
    loading: false,
  }),

  getters: {
    currentSession: (state) =>
      state.sessions.find(s => s.id === state.currentSessionId),
  },

  actions: {
    async sendMessage(content: string) {
      this.loading = true

      // 添加用户消息
      this.messages.push({
        role: 'user',
        content,
        createdAt: new Date(),
      })

      try {
        // 调用 API
        const response = await chatApi.send({
          sessionId: this.currentSessionId,
          message: content,
          hosts: this.selectedHosts,
        })

        // 添加 AI 回复
        this.messages.push({
          role: 'assistant',
          content: response.reply,
          toolCalls: response.toolCalls,
          createdAt: new Date(),
        })

        this.currentSessionId = response.sessionId
      } finally {
        this.loading = false
      }
    },

    async loadSessions() { ... },
    async loadMessages(sessionId: string) { ... },
    setSelectedHosts(hosts: string[]) { ... },
  },
})
```

## API 封装

```typescript
// api/chat.ts
export const chatApi = {
  send(data: { sessionId?: string; message: string; hosts?: string[] }) {
    return request.post('/api/chat', data)
  },

  getSessions() {
    return request.get('/api/chat/sessions')
  },

  getHistory(sessionId: string) {
    return request.get(`/api/chat/history?session_id=${sessionId}`)
  },

  // 流式对话
  sendStream(data: any, onMessage: (event: any) => void) {
    const eventSource = new EventSource('/api/chat/stream')
    eventSource.onmessage = onMessage
    return eventSource
  },
}

// api/host.ts
export const hostApi = {
  list(filter?: HostFilter) {
    return request.get('/api/hosts', { params: filter })
  },

  create(host: Host) {
    return request.post('/api/hosts', host)
  },

  update(id: string, host: Host) {
    return request.put(`/api/hosts/${id}`, host)
  },

  delete(id: string) {
    return request.delete(`/api/hosts/${id}`)
  },

  test(id: string) {
    return request.post(`/api/hosts/${id}/test`)
  },

  import(data: string, format: string) {
    return request.post('/api/hosts/import', { data, format })
  },
}
```

## 交互细节

### 1. 流式输出

```typescript
// 打字机效果
async function streamChat(message: string) {
  const response = await fetch('/api/chat/stream', {
    method: 'POST',
    body: JSON.stringify({ message, sessionId, hosts }),
  })

  const reader = response.body.getReader()
  const decoder = new TextDecoder()

  while (true) {
    const { done, value } = await reader.read()
    if (done) break

    const chunk = decoder.decode(value)
    const lines = chunk.split('\n')

    for (const line of lines) {
      if (line.startsWith('data: ')) {
        const data = JSON.parse(line.slice(6))

        if (data.type === 'thinking') {
          // 显示思考过程
        } else if (data.type === 'tool_call') {
          // 显示工具调用
        } else if (data.type === 'content') {
          // 追加内容
          currentMessage.content += data.content
        }
      }
    }
  }
}
```
